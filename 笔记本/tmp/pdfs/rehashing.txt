        If at most TableSize/2 positions are taken, then an empty spot can

bool RobinHood_Insert ( ElementType Key, HashTable H )
{
    Position Pos = Hash ( Key, H->TableSize );
    int CurrentDisp = 0;
    ElementType CurrentKey = Key;
    while ( true ) {
         if ( H->TheCells[Pos ].Info != Legitimate ) { /* Empty or Deleted */
               H->TheCells[ Pos ].Info = Legitimate;
               H->TheCells[ Pos ].Element = CurrentKey;
               H->TheCells[ Pos ].Disp = CurrentDisp;
               return true;
         }
         if ( CurrentDisp > H->TheCells[ Pos ].Disp ) { /* Rob the rich */
               swap CurrentKey and H->TheCells[ Pos ].Element
               swap CurrentDisp and H->TheCells[ Pos ].Disp
         }
         Pos = ( Pos + 1 ) % H->TableSize ;
         if ( ++CurrentDisp == H->TableSize ) return false; /* table is full */
    }
}
